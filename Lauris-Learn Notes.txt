-- ── reset_demo_passwords.sql ─────────────────────────────────────────────────
--
-- Sets the password for all @lauris.demo accounts to: Demo2026!
--
-- Safe to run multiple times.
-- Run in Supabase SQL editor (service-role context).
-- ─────────────────────────────────────────────────────────────────────────────

DO $$
DECLARE
  _enc TEXT := crypt('Demo2026!', gen_salt('bf', 10));
  _n   INT;
BEGIN
  UPDATE auth.users
     SET encrypted_password = _enc,
         updated_at         = NOW()
   WHERE email LIKE '%@lauris.demo';

  GET DIAGNOSTICS _n = ROW_COUNT;
  RAISE NOTICE 'Password reset to Demo2026! for % @lauris.demo account(s).', _n;
END $$;

SELECT
  u.email,
  p.full_name,
  p.role           AS learn_role,
  p.school_id,
  om.role          AS care_role,
  org.name         AS clinic_name
FROM auth.users u
LEFT JOIN profiles p              ON p.id = u.id
LEFT JOIN organization_memberships om ON om.profile_id = u.id AND om.status = 'active'
LEFT JOIN organizations org       ON org.id = om.organization_id
WHERE u.email LIKE '%@lauris.demo'
ORDER BY org.name, om.role, u.email;
